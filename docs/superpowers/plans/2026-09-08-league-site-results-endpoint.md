```php
<?php
declare(strict_types=1);

/**
 * Results intake from the Reborn backend.
 *
 * The backend knows WHO played, from Discord. This library turns two Discord ids into
 * two clubs and one fixture EXACTLY, then hands the capture to the importer that already
 * knows how to place a slot, resolve a roster, write stat_events and hold for review.
 * No club is ever guessed from a 14-character name here; that fallback lives in
 * si_resolve() for captures that arrive without identity.
 *
 * Every answer is one of: applied, duplicate, review (all ok:true; the backend never
 * resends any of them), or ok:false with code 400 for a body that is not a result.
 */

require_once __DIR__ . '/db.php';
require_once __DIR__ . '/nsr_data.php';
require_once __DIR__ . '/stat_import.php';
require_once __DIR__ . '/nsr_owners.php';

/** league_owners.discord_user_id, added live by __set_owner_discord.php; created here if a replica lacks it. */
function nri_ensure_owner_discord_column(PDO $pdo): void
{
  static $done = false;
  if ($done) return;
  $done = true;
  nsr_owners_schema($pdo);
  $has = (int)$pdo->query("SELECT COUNT(*) FROM information_schema.COLUMNS
                            WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'league_owners'
                              AND COLUMN_NAME = 'discord_user_id'")->fetchColumn();
  if ($has === 0) $pdo->exec("ALTER TABLE league_owners ADD COLUMN discord_user_id VARCHAR(32) NULL");
}

/** Every club an owner runs, by Discord id. Hidden leagues excluded. */
function nri_owner_clubs(PDO $pdo, string $discordId): array
{
  $hidden = NSR_HIDDEN_LEAGUES ? implode(',', array_map('intval', NSR_HIDDEN_LEAGUES)) : '0';
  $st = $pdo->prepare("SELECT o.id AS owner_id, o.handle, ot.league_id, ot.team_id
                         FROM league_owners o JOIN owner_teams ot ON ot.owner_id = o.id
                        WHERE o.discord_user_id = ? AND ot.league_id NOT IN ($hidden)");
  $st->execute([$discordId]);
  return $st->fetchAll(PDO::FETCH_ASSOC);
}

/** INSERT a stat_imports row; a duplicate-key collision answers with the existing row (the July 2026 race). */
function nri_insert_import(PDO $pdo, array $params, bool $ownTx): array
{
  $ins = $pdo->prepare("
    INSERT INTO stat_imports
      (league_id, game_id, series_game_no, capture_hash, status, reason,
       home_team_id, away_team_id, home_score, away_score, captured_at, payload)
    VALUES (:lid,:gid,:sg,:h,:st,:why,:ht,:at,:hs,:as,:ts,:pl)");
  try {
    $ins->execute($params);
    return ['inserted' => true, 'id' => (int)$pdo->lastInsertId()];
  } catch (PDOException $e) {
    if ((int)($e->errorInfo[1] ?? 0) !== 1062) throw $e;
  }
  if ($ownTx && $pdo->inTransaction()) $pdo->rollBack();
  $q = $pdo->prepare("SELECT id, status FROM stat_imports WHERE league_id = :lid AND capture_hash = :h");
  $q->execute(['lid' => $params['lid'], 'h' => $params['h']]);
  $r = $q->fetch(PDO::FETCH_ASSOC) ?: [];
  return ['inserted' => false, 'id' => (int)($r['id'] ?? 0), 'status' => (string)($r['status'] ?? '')];
}

function nri_process(PDO $pdo, array $in): array
{
  si_ensure_schema($pdo);
  nri_ensure_owner_discord_column($pdo);

  if (($in['schema'] ?? '') !== 'nsr.league_result.v1') return ['ok' => false, 'error' => 'Unsupported schema', 'code' => 400];
  $cap = $in['capture'] ?? null;
  if (!is_array($cap) || ($cap['schema'] ?? '') !== 'nsr.game.v1') return ['ok' => false, 'error' => 'Capture is not nsr.game.v1', 'code' => 400];
  if (empty($cap['players']) || !is_array($cap['players'])) return ['ok' => false, 'error' => 'Capture has no players', 'code' => 400];
  if (empty($cap['teams']) || count((array)$cap['teams']) !== 2) return ['ok' => false, 'error' => 'Capture has no teams', 'code' => 400];

  $humans = [];
  foreach ((array)($in['participants'] ?? []) as $p) {
    if (($p['kind'] ?? 'human') !== 'human') continue;
    $id = trim((string)($p['discord_user_id'] ?? ''));
    if ($id === '') continue;
    $humans[] = ['discord' => $id, 'team_index' => isset($p['team_index']) && in_array((int)$p['team_index'], [0, 1], true) ? (int)$p['team_index'] : null];
  }

  $ts = null;
  if (!empty($in['captured_at']) && ($t = strtotime((string)$in['captured_at'])) !== false) $ts = date('Y-m-d H:i:s', $t);
  if ($ts === null) $ts = gmdate('Y-m-d H:i:s');
  $raw = json_encode($in, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
  $hash = si_capture_hash($cap);
  $ownTx = !$pdo->inTransaction();

  /* Hold with a reason, without a league: a row a commissioner can act on. league_id 0 marks
     "no league resolved"; the dashboard lists it under its own heading. */
  $hold = function (int $leagueId, ?array $ctx, string $reason) use ($pdo, $hash, $ts, $raw, $ownTx): array {
    $r = nri_insert_import($pdo, [
      'lid' => $leagueId, 'gid' => $ctx['game_id'] ?? null, 'sg' => $ctx['series_game_no'] ?? null,
      'h' => $hash, 'st' => 'review', 'why' => mb_substr($reason, 0, 255),
      'ht' => $ctx['home_team_id'] ?? null, 'at' => $ctx['away_team_id'] ?? null,
      'hs' => $ctx['home_score'] ?? null, 'as' => $ctx['away_score'] ?? null, 'ts' => $ts, 'pl' => $raw,
    ], $ownTx);
    if (!$r['inserted']) return ['ok' => true, 'status' => 'duplicate', 'import_id' => $r['id'], 'existing_status' => $r['status']];
    return ['ok' => true, 'status' => 'review', 'import_id' => $r['id'], 'reason' => $reason];
  };

  if (count($humans) < 2) return $hold(0, null, 'needs two signed-in players; got ' . count($humans));

  $clubs = [];
  foreach ($humans as $h) {
    $rows = nri_owner_clubs($pdo, $h['discord']);
    if (!$rows) return $hold(0, null, 'no owner linked to Discord id ' . $h['discord'] . ' - link it in the owners page');
    $clubs[] = $rows;
  }

  /* Leagues where both own a club and an incomplete fixture between those two clubs exists. */
  $candidates = [];
  foreach ($clubs[0] as $a) {
    foreach ($clubs[1] as $b) {
      if ((int)$a['league_id'] !== (int)$b['league_id']) continue;
      $lid = (int)$a['league_id'];
      foreach (si_fixtures($pdo, $lid, (int)$a['team_id'], (int)$b['team_id']) as $fx) {
        if (si_next_slot($pdo, $lid, (int)$fx['id']) === 0) continue;   // a finished best-of-3 never absorbs a later game
        $candidates[] = ['league_id' => $lid, 'fixture' => $fx, 'a' => $a, 'b' => $b];
      }
    }
  }
  if (!$candidates) {
    $ha = (string)($clubs[0][0]['handle'] ?? $humans[0]['discord']);
    $hb = (string)($clubs[1][0]['handle'] ?? $humans[1]['discord']);
    return $hold(0, null, "no incomplete fixture between {$ha} and {$hb} in any league");
  }
  if (count($candidates) > 1) {
    $list = array_map(fn($c) => "league {$c['league_id']} week {$c['fixture']['week_num']}", $candidates);
    return $hold((int)$candidates[0]['league_id'], ['home_team_id' => (int)$candidates[0]['fixture']['home_team_id'], 'away_team_id' => (int)$candidates[0]['fixture']['away_team_id']],
                 count($candidates) . ' fixtures match: ' . implode(', ', $list) . ' - pick one');
  }
  $chosen = $candidates[0];
  $leagueId = (int)$chosen['league_id'];

  $dup = $pdo->prepare("SELECT id, status FROM stat_imports WHERE league_id = :lid AND capture_hash = :h");
  $dup->execute(['lid' => $leagueId, 'h' => $hash]);
  if ($row = $dup->fetch(PDO::FETCH_ASSOC)) return ['ok' => true, 'status' => 'duplicate', 'import_id' => (int)$row['id'], 'existing_status' => (string)$row['status']];

  /* Which capture team is which club: from the backend's team_index when both are known and differ. */
  $teamIds = null;
  $ia = $humans[0]['team_index']; $ib = $humans[1]['team_index'];
  if ($ia !== null && $ib !== null && $ia !== $ib) {
    $teamIds = [$ia => (int)$chosen['a']['team_id'], $ib => (int)$chosen['b']['team_id']];
  } else {
    /* Fall back to the capture's names, but only against these two clubs. */
    $tq = $pdo->prepare("SELECT id, name FROM teams WHERE id IN (?, ?)");
    $tq->execute([(int)$chosen['a']['team_id'], (int)$chosen['b']['team_id']]);
    $two = $tq->fetchAll(PDO::FETCH_ASSOC);
    $t0 = si_resolve_team($two, (string)($cap['teams'][0]['name'] ?? ''));
    $t1 = si_resolve_team($two, (string)($cap['teams'][1]['name'] ?? ''));
    if ($t0 && $t1 && $t0['id'] !== $t1['id']) $teamIds = [0 => $t0['id'], 1 => $t1['id']];
  }
  if ($teamIds === null) {
    return $hold($leagueId, ['home_team_id' => (int)$chosen['fixture']['home_team_id'], 'away_team_id' => (int)$chosen['fixture']['away_team_id']],
                 'could not tell which capture team is which club (no team_index and names did not settle it)');
  }

  $res = si_resolve($pdo, $leagueId, $cap, ['game_id' => (int)$chosen['fixture']['id'], 'team_ids' => $teamIds]);
  $ctx = $res['ctx'] ?? [];
  if (!$res['ok']) return $hold($leagueId, $ctx, (string)$res['reason']);

  try {
    if ($ownTx) $pdo->beginTransaction();
    $ins = nri_insert_import($pdo, [
      'lid' => $leagueId, 'gid' => (int)$ctx['game_id'], 'sg' => (int)$ctx['series_game_no'],
      'h' => $hash, 'st' => 'applied', 'why' => null,
      'ht' => (int)$ctx['home_team_id'], 'at' => (int)$ctx['away_team_id'],
      'hs' => (int)$ctx['home_score'], 'as' => (int)$ctx['away_score'], 'ts' => $ts, 'pl' => $raw,
    ], $ownTx);
    if (!$ins['inserted']) return ['ok' => true, 'status' => 'duplicate', 'import_id' => $ins['id'], 'existing_status' => $ins['status']];
    $written = si_apply($pdo, $leagueId, $ins['id'], $ctx);
    if ($ownTx) $pdo->commit();
  } catch (Throwable $e) {
    if ($ownTx && $pdo->inTransaction()) $pdo->rollBack();
    throw $e;
  }
  return ['ok' => true, 'status' => 'applied', 'import_id' => $ins['id'], 'league_id' => $leagueId,
          'game_id' => (int)$ctx['game_id'], 'series_game_no' => (int)$ctx['series_game_no'], 'events' => $written];
}
```

```php
<?php
declare(strict_types=1);

/**
 * League results, from the Reborn backend only.
 *
 * Players never call this. The backend has already verified WHO played through Discord,
 * and it presents its own bearer token (BACKEND_TOKEN, separate from the bot's). The body
 * is nsr.league_result.v1: the capture plus both participants' Discord ids.
 *
 * Answers, always JSON:
 *   {"ok":true,"status":"applied","import_id":N,"game_id":N,"series_game_no":N}
 *   {"ok":true,"status":"duplicate","import_id":N}
 *   {"ok":true,"status":"review","import_id":N,"reason":"..."}
 *   {"ok":false,"error":"..."}
 * review is a SUCCESS for the caller: stored, a commissioner decides, never resend.
 */

require_once __DIR__ . '/../_inc/config.php';
require_once __DIR__ . '/../_inc/db.php';
require_once __DIR__ . '/../_inc/nsr_results_intake.php';

header('Content-Type: application/json; charset=utf-8');
function rfb_out(array $d): void { echo json_encode($d, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE); exit; }
function rfb_fail(string $msg, int $code = 400): void { http_response_code($code); rfb_out(['ok' => false, 'error' => $msg]); }

$tokenFile = __DIR__ . '/../_inc/backend_token.php';
if (!is_file($tokenFile)) rfb_fail('Intake not configured', 503);
require_once $tokenFile;
if (!defined('BACKEND_TOKEN') || BACKEND_TOKEN === '') rfb_fail('Intake not configured', 503);

$hdr = '';
foreach (['HTTP_AUTHORIZATION', 'REDIRECT_HTTP_AUTHORIZATION'] as $k) {
  if (!empty($_SERVER[$k])) { $hdr = (string)$_SERVER[$k]; break; }
}
if ($hdr === '' && function_exists('apache_request_headers')) {
  foreach (apache_request_headers() as $k => $v) {
    if (strcasecmp($k, 'Authorization') === 0) { $hdr = (string)$v; break; }
  }
}
$sent = '';
if (preg_match('/Bearer\s+(.+)/i', $hdr, $m)) $sent = trim($m[1]);
if ($sent === '' || !hash_equals(BACKEND_TOKEN, $sent)) rfb_fail('Unauthorized', 401);
if (($_SERVER['REQUEST_METHOD'] ?? '') !== 'POST') rfb_fail('POST required', 405);

$raw = file_get_contents('php://input') ?: '';
if ($raw === '') rfb_fail('Empty body');
if (strlen($raw) > 512 * 1024) rfb_fail('Payload too large', 413);
$in = json_decode($raw, true);
if (!is_array($in)) rfb_fail('Body is not JSON');

try {
  $r = nri_process(db(), $in);
} catch (Throwable $e) {
  rfb_fail('Intake failed: ' . $e->getMessage(), 500);
}
if (($r['ok'] ?? false) !== true) rfb_fail((string)($r['error'] ?? 'Bad request'), (int)($r['code'] ?? 400));
rfb_out($r);
```
